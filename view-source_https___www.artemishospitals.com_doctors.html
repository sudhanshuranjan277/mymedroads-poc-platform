dataType: "json",
success: function (response) {

// -------------------------------
// MENU LINKS FILTERING
// -------------------------------

return item.title.toLowerCase().includes(query.toLowerCase());
}).map(function (item) {
return "<a href='" + item.url + "'>" + item.title + "</a>";
});

// -------------------------------
// FINAL MERGE API + MENU LINKS
// -------------------------------
var finalResults = apiResults.concat(menuResults);

// SHOW RESULT IN DIV 1
var s1 = $("#suggestionsheader");
s1.empty();
if (finalResults.length > 0) {
finalResults.forEach(function (item) {
s1.append(item + "<br/>");

s1.hide();

var s2 = $("#suggestionsheader2");
s2.empty();
if (finalResults.length > 0) {
finalResults.forEach(function (item) {
s2.append(item + "<br/>");

s2.hide();
}
}
});
}
